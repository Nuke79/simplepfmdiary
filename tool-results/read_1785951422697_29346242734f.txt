    50→/* ---------- types ---------- */
    51→interface Measurement {
    52→  id: string;
    53→  value: number;
    54→  period: "morning" | "evening";
    55→  timing: "before" | "after";
    56→  date: string; // ISO string
    57→  createdAt: string;
    58→}
    59→
    60→interface Settings {
    61→  id: string;
    62→  personalBest: number;
    63→}
    64→
    65→/* ---------- zone helpers ---------- */
    66→function getZone(value: number, pb: number): "green" | "yellow" | "red" {
    67→  if (value >= pb * 0.8) return "green";
    68→  if (value >= pb * 0.5) return "yellow";
    69→  return "red";
    70→}
    71→
    72→function zoneColor(zone: "green" | "yellow" | "red") {
    73→  if (zone === "green") return "bg-emerald-500";
    74→  if (zone === "yellow") return "bg-amber-500";
    75→  return "bg-red-500";
    76→}
    77→
    78→function zoneLabel(zone: "green" | "yellow" | "red") {
    79→  if (zone === "green") return "Зелёная";
    80→  if (zone === "yellow") return "Жёлтая";
    81→  return "Красная";
    82→}
    83→
    84→function zoneBorderColor(zone: "green" | "yellow" | "red") {
    85→  if (zone === "green") return "border-emerald-500";
    86→  if (zone === "yellow") return "border-amber-500";
    87→  return "border-red-500";
    88→}
    89→
    90→/* ---------- chart config ---------- */
    91→const chartConfig = {
    92→  morningBefore: { label: "Утро до инг.", color: "#f97316" },
    93→  morningAfter: { label: "Утро после инг.", color: "#ea580c" },
    94→  eveningBefore: { label: "Вечер до инг.", color: "#8b5cf6" },
    95→  eveningAfter: { label: "Вечер после инг.", color: "#7c3aed" },
    96→} satisfies ChartConfig;
    97→
    98→/* ---------- main component ---------- */
    99→export default function PeakFlowDiary() {
   100→  const [measurements, setMeasurements] = useState<Measurement[]>([]);
   101→  const [settings, setSettings] = useState<Settings>({ id: "singleton", personalBest: 400 });
   102→  const [inputValue, setInputValue] = useState("");
   103→  const [period, setPeriod] = useState<"morning" | "evening">(() => {
   104→    const hour = new Date().getHours();
   105→    return hour >= 5 && hour < 15 ? "morning" : "evening";
   106→  });
   107→  const [timing, setTiming] = useState<"before" | "after">("before");
   108→  const [pbInput, setPbInput] = useState("");
   109→  const [notificationsEnabled, setNotificationsEnabled] = useState(false);
   110→  const [reminderTimeout, setReminderTimeout] = useState<NodeJS.Timeout | null>(null);
   111→  const [editingPB, setEditingPB] = useState(false);
   112→  const inputRef = useRef<HTMLInputElement>(null);
   113→  const [chartDays, setChartDays] = useState(14);
   114→
   115→  /* --- fetch data --- */
   116→  useEffect(() => {
   117→    const controller = new AbortController();
   118→    const signal = controller.signal;
   119→
   120→    (async () => {
   121→      try {
   122→        const from = subDays(new Date(), chartDays).toISOString();
   123→        const mRes = await fetch(`/api/measurements?from=${from}`, { signal });
   124→        if (mRes.ok && !signal.aborted) setMeasurements(await mRes.json());
   125→
   126→        const sRes = await fetch("/api/settings", { signal });
   127→        if (sRes.ok && !signal.aborted) {
   128→          const s = await sRes.json();
   129→          setSettings(s);
   130→          setPbInput(String(s.personalBest));
   131→        }
   132→      } catch { /* aborted */ }
   133→    })();
   134→
   135→    return () => controller.abort();
   136→  }, [chartDays]);
   137→
   138→  /* --- add measurement --- */
   139→  const addMeasurement = async () => {
   140→    const val = parseInt(inputValue);
   141→    if (isNaN(val) || val < 50 || val > 900) {
   142→      toast.error("Введите корректное значение ПСВ (50-900 л/мин)");
   143→      return;
   144→    }
   145→
   146→    const now = new Date();
   147→    await fetch("/api/measurements", {
   148→      method: "POST",
   149→      headers: { "Content-Type": "application/json" },
   150→      body: JSON.stringify({
   151→        value: val,
   152→        period,
   153→        timing,
   154→        date: now.toISOString(),
   155→      }),
   156→    });
   157→
   158→    setInputValue("");
   159→
   160→    // If this is a "before" measurement, schedule a 30-min reminder
   161→    if (timing === "before") {
   162→      setTiming("after");
   163→      scheduleReminder();
   164→    } else {
   165→      // After inhaler measurement done, keep timing as "before" for next session
   166→      setTiming("before");
   167→    }
   168→
   169→    toast.success(`Записано: ${val} л/мин (${period === "morning" ? "утро" : "вечер"}, ${timing === "before" ? "до" : "после"} ингаляции)`);
   170→    // Refetch measurements
   171→    const from = subDays(new Date(), chartDays).toISOString();
   172→    const mRes = await fetch(`/api/measurements?from=${from}`);
   173→    if (mRes.ok) setMeasurements(await mRes.json());
   174→    inputRef.current?.focus();
   175→  };
   176→
   177→  /* --- delete measurement --- */
   178→  const deleteMeasurement = async (id: string) => {
   179→    await fetch(`/api/measurements?id=${id}`, { method: "DELETE" });
   180→    const from = subDays(new Date(), chartDays).toISOString();
   181→    const mRes = await fetch(`/api/measurements?from=${from}`);
   182→    if (mRes.ok) setMeasurements(await mRes.json());
   183→    toast.success("Запись удалена");
   184→  };
   185→
   186→  /* --- update personal best --- */
   187→  const savePB = async () => {
   188→    const val = parseInt(pbInput);
   189→    if (isNaN(val) || val < 50 || val > 900) {
   190→      toast.error("Введите корректное значение (50-900)");
   191→      return;
   192→    }
   193→    await fetch("/api/settings", {
   194→      method: "PUT",
   195→      headers: { "Content-Type": "application/json" },
   196→      body: JSON.stringify({ personalBest: val }),
   197→    });
   198→    setSettings({ ...settings, personalBest: val });
   199→    setEditingPB(false);
   200→    toast.success(`Персональный лучший результат: ${val} л/мин`);
   201→  };
   202→
   203→  /* --- notifications --- */
   204→  const scheduleReminder = () => {
   205→    if (!notificationsEnabled) return;
   206→    if (reminderTimeout) clearTimeout(reminderTimeout);
   207→    const timeout = setTimeout(() => {
   208→      if (Notification.permission === "granted") {
   209→        new Notification("Пикфлоуметрия", {
   210→          body: "Прошло 30 минут — время сделать замер после ингаляции!",
   211→          icon: "/logo.svg",
   212→          tag: "peakflow-reminder",
   213→        });
   214→      }
   215→      toast.info("Прошло 30 минут — время сделать замер после ингаляции!", {
   216→        duration: 10000,
   217→      });
   218→    }, 30 * 60 * 1000); // 30 minutes
   219→    setReminderTimeout(timeout);
   220→  };
   221→
   222→  const toggleNotifications = async () => {
   223→    if (!notificationsEnabled) {
   224→      if ("Notification" in window) {
   225→        const perm = await Notification.requestPermission();
   226→        if (perm === "granted") {
   227→          setNotificationsEnabled(true);
   228→          toast.success("Уведомления включены");
   229→        } else {
   230→          toast.error("Разрешите уведомления в браузере");
   231→        }
   232→      } else {
   233→        toast.error("Уведомления не поддерживаются в этом браузере");
   234→      }
   235→    } else {
   236→      setNotificationsEnabled(false);
   237→      if (reminderTimeout) clearTimeout(reminderTimeout);
   238→      toast.success("Уведомления выключены");
   239→    }
   240→  };
   241→
   242→  /* --- CSV export --- */
   243→  const exportCSV = () => {
   244→    const from = subDays(new Date(), chartDays).toISOString();
   245→    window.open(`/api/export?from=${from}`, "_blank");
   246→    toast.success("CSV-файл загружен");
   247→  };
   248→
   249→  /* --- chart data --- */
   250→  const chartData = React.useMemo(() => {
   251→    const days = Array.from({ length: chartDays }, (_, i) => {
   252→      const d = startOfDay(subDays(new Date(), chartDays - 1 - i));
   253→      const dayStr = format(d, "yyyy-MM-dd");
   254→      const dayMs = [
   255→        { key: "morningBefore", period: "morning", timing: "before" },
   256→        { key: "morningAfter", period: "morning", timing: "after" },
   257→        { key: "eveningBefore", period: "evening", timing: "before" },
   258→        { key: "eveningAfter", period: "evening", timing: "after" },
   259→      ] as const;
   260→
   261→      const point: Record<string, string | number> = { date: format(d, "dd.MM", { locale: ru }) };
   262→      for (const { key, period: p, timing: t } of dayMs) {
   263→        const m = measurements.find(
   264→          (m) =>
   265→            format(new Date(m.date), "yyyy-MM-dd") === dayStr &&
   266→            m.period === p &&
   267→            m.timing === t
   268→        );
   269→        point[key] = m ? m.value : undefined as unknown as number;
   270→      }
   271→      return point;
   272→    });
   273→    return days;
   274→  }, [measurements, chartDays]);
   275→
   276→  const greenMin = Math.round(settings.personalBest * 0.8);
   277→  const yellowMin = Math.round(settings.personalBest * 0.5);
   278→
   279→  /* --- today's measurements --- */
   280→  const todayMeasurements = React.useMemo(() => {
   281→    const todayStr = format(new Date(), "yyyy-MM-dd");
   282→    return measurements.filter(
   283→      (m) => format(new Date(m.date), "yyyy-MM-dd") === todayStr
   284→    );
   285→  }, [measurements]);
   286→
   287→  const todayDone = (p: "morning" | "evening", t: "before" | "after") =>
   288→    todayMeasurements.some((m) => m.period === p && m.timing === t);
   289→
   290→  /* --- history list --- */
   291→  const historyList = React.useMemo(() => {
   292→    return [...measurements]
   293→      .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   294→      .slice(0, 20);
   295→  }, [measurements]);
   296→
   297→  return (
   298→    <div className="min-h-screen bg-gradient-to-b from-slate-50 to-slate-100 flex flex-col">
   299→      {/* Header */}
   300→      <header className="bg-white/80 backdrop-blur-sm border-b border-slate-200 sticky top-0 z-10">
   301→        <div className="max-w-lg mx-auto px-4 py-3 flex items-center justify-between">
   302→          <div className="flex items-center gap-2">
   303→            <Wind className="h-6 w-6 text-emerald-600" />
   304→            <h1 className="text-lg font-bold text-slate-800">Пикфлоуметрия</h1>
   305→          </div>
   306→          <div className="flex items-center gap-2">
   307→            <Button
   308→              variant="ghost"
   309→              size="icon"
   310→              onClick={toggleNotifications}
   311→              className="h-9 w-9"
   312→              title={notificationsEnabled ? "Выключить уведомления" : "Включить уведомления"}
   313→            >
   314→              {notificationsEnabled ? (
   315→                <Bell className="h-4 w-4 text-emerald-600" />
   316→              ) : (
   317→                <BellOff className="h-4 w-4 text-slate-400" />
   318→              )}
   319→            </Button>
   320→            <Button
   321→              variant="ghost"
   322→              size="icon"
   323→              onClick={exportCSV}
   324→              className="h-9 w-9"
   325→              title="Экспорт CSV"
   326→            >
   327→              <Download className="h-4 w-4 text-slate-600" />
   328→            </Button>
   329→          </div>
   330→        </div>
   331→      </header>
   332→
   333→      {/* Main content */}
   334→      <main className="flex-1 max-w-lg mx-auto w-full px-4 py-4 pb-8">
   335→        <Tabs defaultValue="input" className="w-full">
   336→          <TabsList className="w-full grid grid-cols-3">
   337→            <TabsTrigger value="input" className="gap-1.5">
   338→              <Plus className="h-3.5 w-3.5" />
   339→              <span className="text-xs sm:text-sm">Ввод</span>
   340→            </TabsTrigger>
   341→            <TabsTrigger value="chart" className="gap-1.5">
   342→              <Activity className="h-3.5 w-3.5" />
   343→              <span className="text-xs sm:text-sm">График</span>
   344→            </TabsTrigger>
   345→            <TabsTrigger value="history" className="gap-1.5">
   346→              <Clock className="h-3.5 w-3.5" />
   347→              <span className="text-xs sm:text-sm">История</span>
   348→            </TabsTrigger>
   349→          </TabsList>
   350→
   351→          {/* ===== INPUT TAB ===== */}
   352→          <TabsContent value="input">
   353→            <div className="space-y-4 mt-4">
   354→              {/* Today's checklist */}
   355→              <Card>
   356→                <CardHeader className="pb-2">
   357→                  <CardTitle className="text-sm font-medium flex items-center gap-2">
   358→                    <Check className="h-4 w-4 text-emerald-500" />
   359→                    Сегодня, {format(new Date(), "dd MMMM", { locale: ru })}
   360→                  </CardTitle>
   361→                </CardHeader>
   362→                <CardContent>
   363→                  <div className="grid grid-cols-2 gap-2">
   364→                    {(
   365→                      [
   366→                        { p: "morning" as const, t: "before" as const, label: "Утро (до)", icon: Sun },
   367→                        { p: "morning" as const, t: "after" as const, label: "Утро (после)", icon: Sun },
   368→                        { p: "evening" as const, t: "before" as const, label: "Вечер (до)", icon: Moon },
   369→                        { p: "evening" as const, t: "after" as const, label: "Вечер (после)", icon: Moon },
   370→                      ] as const
   371→                    ).map(({ p, t, label, icon: Icon }) => (
   372→                      <div
   373→                        key={`${p}-${t}`}
   374→                        className={`flex items-center gap-2 rounded-lg border p-2.5 text-sm transition-colors ${
   375→                          todayDone(p, t)
   376→                            ? "bg-emerald-50 border-emerald-200 text-emerald-700"
   377→                            : "bg-slate-50 border-slate-200 text-slate-500"
   378→                        } ${period === p && timing === t ? "ring-2 ring-emerald-500 ring-offset-1" : ""}`}
   379→                      >
   380→                        <Icon className="h-4 w-4 shrink-0" />
   381→                        <span className="text-xs font-medium leading-tight">{label}</span>
   382→                        {todayDone(p, t) && (
   383→                          <Check className="h-3.5 w-3.5 text-emerald-500 ml-auto shrink-0" />
   384→                        )}
   385→                      </div>
   386→                    ))}
   387→                  </div>
   388→                </CardContent>
   389→              </Card>
   390→
   391→              {/* Input card */}
   392→              <Card className="border-2 border-emerald-200">
   393→                <CardHeader className="pb-2">
   394→                  <CardTitle className="text-base font-semibold">Новый замер</CardTitle>
   395→                  <CardDescription>
   396→                    {period === "morning" ? (
   397→                      <span className="flex items-center gap-1">
   398→                        <Sun className="h-3.5 w-3.5 text-amber-500" /> Утро
   399→                      </span>
   400→                    ) : (
   401→                      <span className="flex items-center gap-1">
   402→                        <Moon className="h-3.5 w-3.5 text-indigo-400" /> Вечер
   403→                      </span>
   404→                    )}
   405→                    {" — "}
   406→                    {timing === "before" ? "до ингаляции" : "через 30 мин после ингаляции"}
   407→                  </CardDescription>
   408→                </CardHeader>
   409→                <CardContent className="space-y-4">
   410→                  {/* Period toggle */}
   411→                  <div className="flex gap-2">
   412→                    <Button
   413→                      variant={period === "morning" ? "default" : "outline"}
   414→                      size="sm"
   415→                      className="flex-1 gap-1.5"
   416→                      onClick={() => setPeriod("morning")}
   417→                    >
   418→                      <Sun className="h-3.5 w-3.5" /> Утро
   419→                    </Button>
   420→                    <Button
   421→                      variant={period === "evening" ? "default" : "outline"}
   422→                      size="sm"
   423→                      className="flex-1 gap-1.5"
   424→                      onClick={() => setPeriod("evening")}
   425→                    >
   426→                      <Moon className="h-3.5 w-3.5" /> Вечер
   427→                    </Button>
   428→                  </div>
   429→
   430→                  {/* Timing toggle */}
   431→                  <div className="flex gap-2">
   432→                    <Button
   433→                      variant={timing === "before" ? "default" : "outline"}
   434→                      size="sm"
   435→                      className="flex-1"
   436→                      onClick={() => setTiming("before")}
   437→                    >
   438→                      До ингаляции
   439→                    </Button>
   440→                    <Button
   441→                      variant={timing === "after" ? "default" : "outline"}
   442→                      size="sm"
   443→                      className="flex-1"
   444→                      onClick={() => setTiming("after")}
   445→                    >
   446→                      После ингаляции
   447→                    </Button>
   448→                  </div>
   449→
   450→                  {/* Value input */}
   451→                  <div className="flex gap-2 items-end">
   452→                    <div className="flex-1">
   453→                      <label className="text-xs text-muted-foreground mb-1 block">
   454→                        ПСВ (л/мин)
   455→                      </label>
   456→                      <Input
   457→                        ref={inputRef}
   458→                        type="number"
   459→                        inputMode="numeric"
   460→                        placeholder="250"
   461→                        value={inputValue}
   462→                        onChange={(e) => setInputValue(e.target.value)}
   463→                        onKeyDown={(e) => {
   464→                          if (e.key === "Enter") addMeasurement();
   465→                        }}
   466→                        min={50}
   467→                        max={900}
   468→                        className="text-2xl font-bold h-14 text-center"
   469→                      />
   470→                    </div>
   471→                    <Button
   472→                      size="lg"
   473→                      className="h-14 px-6 text-base font-semibold bg-emerald-600 hover:bg-emerald-700"
   474→                      onClick={addMeasurement}
   475→                    >
   476→                      <Plus className="h-5 w-5 mr-1" />
   477→                      Записать
   478→                    </Button>
   479→                  </div>
   480→
   481→                  {/* Zone indicator */}
   482→                  {inputValue && !isNaN(parseInt(inputValue)) && (
   483→                    <div
   484→                      className={`flex items-center justify-center gap-2 rounded-lg p-3 border-l-4 ${zoneBorderColor(
   485→                        getZone(parseInt(inputValue), settings.personalBest)
   486→                      )} ${
   487→                        getZone(parseInt(inputValue), settings.personalBest) === "green"
   488→                          ? "bg-emerald-50"
   489→                          : getZone(parseInt(inputValue), settings.personalBest) === "yellow"
   490→                          ? "bg-amber-50"
   491→                          : "bg-red-50"
   492→                      }`}
   493→                    >
   494→                      <div className={`h-3 w-3 rounded-full ${zoneColor(getZone(parseInt(inputValue), settings.personalBest))}`} />
   495→                      <span className="text-sm font-medium">
   496→                        {zoneLabel(getZone(parseInt(inputValue), settings.personalBest))} зона
   497→                      </span>
   498→                      <span className="text-xs text-muted-foreground ml-1">
   499→                        ({parseInt(inputValue)} / {settings.personalBest})
   500→                      </span>
   501→                    </div>
   502→                  )}
   503→                </CardContent>
   504→              </Card>
   505→
   506→              {/* Zone legend */}
   507→              <Card>
   508→                <CardHeader className="pb-2">
   509→                  <CardTitle className="text-sm font-medium flex items-center gap-2">
   510→                    <Settings className="h-4 w-4" /> Зоны и настройки
   511→                  </CardTitle>
   512→                </CardHeader>
   513→                <CardContent className="space-y-3">
   514→                  <div className="space-y-2">
   515→                    <div className="flex items-center justify-between text-sm">
   516→                      <div className="flex items-center gap-2">
   517→                        <div className="h-3 w-3 rounded-full bg-emerald-500" />
   518→                        <span>Зелёная</span>
   519→                      </div>
   520→                      <span className="text-muted-foreground">
   521→                        {greenMin}–{settings.personalBest} л/мин
   522→                      </span>
   523→                    </div>
   524→                    <div className="flex items-center justify-between text-sm">
   525→                      <div className="flex items-center gap-2">
   526→                        <div className="h-3 w-3 rounded-full bg-amber-500" />
   527→                        <span>Жёлтая</span>
   528→                      </div>
   529→                      <span className="text-muted-foreground">
   530→                        {yellowMin}–{greenMin - 1} л/мин
   531→                      </span>
   532→                    </div>
   533→                    <div className="flex items-center justify-between text-sm">
   534→                      <div className="flex items-center gap-2">
   535→                        <div className="h-3 w-3 rounded-full bg-red-500" />
   536→                        <span>Красная</span>
   537→                      </div>
   538→                      <span className="text-muted-foreground">
