    51→/* ---------- types ---------- */
    52→interface Measurement {
    53→  id: string;
    54→  value: number;
    55→  period: "morning" | "evening";
    56→  timing: "before" | "after";
    57→  date: string; // ISO string
    58→  createdAt: string;
    59→}
    60→
    61→interface AppSettings {
    62→  personalBest: number;
    63→}
    64→
    65→/* ---------- local storage helpers ---------- */
    66→const STORAGE_KEYS = {
    67→  measurements: "peakflow_measurements",
    68→  settings: "peakflow_settings",
    69→  notifications: "peakflow_notifications",
    70→} as const;
    71→
    72→function loadMeasurements(): Measurement[] {
    73→  if (typeof window === "undefined") return [];
    74→  try {
    75→    const raw = localStorage.getItem(STORAGE_KEYS.measurements);
    76→    return raw ? JSON.parse(raw) : [];
    77→  } catch {
    78→    return [];
    79→  }
    80→}
    81→
    82→function saveMeasurements(data: Measurement[]) {
    83→  localStorage.setItem(STORAGE_KEYS.measurements, JSON.stringify(data));
    84→}
    85→
    86→function loadSettings(): AppSettings {
    87→  if (typeof window === "undefined") return { personalBest: 400 };
    88→  try {
    89→    const raw = localStorage.getItem(STORAGE_KEYS.settings);
    90→    return raw ? JSON.parse(raw) : { personalBest: 400 };
    91→  } catch {
    92→    return { personalBest: 400 };
    93→  }
    94→}
    95→
    96→function saveSettings(data: AppSettings) {
    97→  localStorage.setItem(STORAGE_KEYS.settings, JSON.stringify(data));
    98→}
    99→
   100→function loadNotificationsPref(): boolean {
   101→  if (typeof window === "undefined") return false;
   102→  return localStorage.getItem(STORAGE_KEYS.notifications) === "true";
   103→}
   104→
   105→function saveNotificationsPref(val: boolean) {
   106→  localStorage.setItem(STORAGE_KEYS.notifications, JSON.stringify(val));
   107→}
   108→
   109→function generateId(): string {
   110→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 8);
   111→}
   112→
   113→/* ---------- zone helpers ---------- */
   114→function round10(n: number): number {
   115→  return Math.round(n / 10) * 10;
   116→}
   117→
   118→function getZone(value: number, pb: number): "green" | "yellow" | "red" {
   119→  if (value >= round10(pb * 0.8)) return "green";
   120→  if (value >= round10(pb * 0.5)) return "yellow";
   121→  return "red";
   122→}
   123→
   124→function zoneColor(zone: "green" | "yellow" | "red") {
   125→  if (zone === "green") return "bg-emerald-500";
   126→  if (zone === "yellow") return "bg-amber-500";
   127→  return "bg-red-500";
   128→}
   129→
   130→function zoneLabel(zone: "green" | "yellow" | "red") {
   131→  if (zone === "green") return "Зелёная";
   132→  if (zone === "yellow") return "Жёлтая";
   133→  return "Красная";
   134→}
   135→
   136→function zoneBorderColor(zone: "green" | "yellow" | "red") {
   137→  if (zone === "green") return "border-emerald-500";
   138→  if (zone === "yellow") return "border-amber-500";
   139→  return "border-red-500";
   140→}

   150→/* ---------- main component ---------- */
   151→export default function PeakFlowDiary() {
   152→  const [mounted, setMounted] = useState(false);
   153→
   154→  const [measurements, setMeasurements] = useState<Measurement[]>([]);
   155→  const [settings, setSettings] = useState<AppSettings>({ personalBest: 400 });
   156→  const [inputValue, setInputValue] = useState("");
   157→  const [period, setPeriod] = useState<"morning" | "evening">("morning");
   158→  const [timing, setTiming] = useState<"before" | "after">("before");
   159→  const [pbInput, setPbInput] = useState("400");
   160→  const [notificationsEnabled, setNotificationsEnabled] = useState(false);
   161→  const [reminderTimeout, setReminderTimeout] = useState<ReturnType<typeof setTimeout> | null>(null);
   162→  const [editingPB, setEditingPB] = useState(false);
   163→  const inputRef = useRef<HTMLInputElement>(null);
   164→  const [chartDays, setChartDays] = useState(14);
   165→
   166→  /* --- load data from localStorage once after mount --- */
   167→  useEffect(() => {
   168→    // eslint-disable-next-line react-hooks/set-state-in-effect -- one-time init from browser API (localStorage)
   169→    setMeasurements(loadMeasurements());
   170→    const s = loadSettings();
   171→    setSettings(s);
   172→    setPbInput(String(s.personalBest));
   173→    setNotificationsEnabled(loadNotificationsPref());
   174→    const hour = new Date().getHours();
   175→    setPeriod(hour >= 5 && hour < 15 ? "morning" : "evening");
   176→    // Pre-fill with last measurement value
   177→    const saved = loadMeasurements();
   178→    if (saved.length > 0) {
   179→      const lastVal = saved[saved.length - 1].value;
   180→      setInputValue(String(lastVal));
   181→    }
   182→    setMounted(true);
   183→  }, []);
   184→
   185→  /* --- add measurement --- */
   186→  const addMeasurement = () => {
   187→    const val = parseInt(inputValue);
   188→    if (isNaN(val) || val < 50 || val > 900) {
   189→      toast.error("Введите корректное значение ПСВ (50-900 л/мин)");
   190→      return;
   191→    }
   192→
   193→    const now = new Date();
   194→    const newM: Measurement = {
   195→      id: generateId(),
   196→      value: val,
   197→      period,
   198→      timing,
   199→      date: now.toISOString(),
   200→      createdAt: now.toISOString(),
   201→    };
   202→
   203→    const updated = [...measurements, newM];
   204→    setMeasurements(updated);
   205→    saveMeasurements(updated);
   206→
   207→    setInputValue(String(val));
   208→
   209→    // If this is a "before" measurement, switch to "after" and schedule reminder
   210→    if (timing === "before") {
   211→      setTiming("after");
   212→      scheduleReminder();
   213→    } else {
   214→      setTiming("before");
   215→    }
   216→
   217→    toast.success(`Записано: ${val} л/мин (${period === "morning" ? "утро" : "вечер"}, ${timing === "before" ? "до" : "после"} ингаляции)`);
   218→    inputRef.current?.focus();
   219→  };
   220→
   221→  /* --- delete measurement --- */
   222→  const deleteMeasurement = (id: string) => {
   223→    const updated = measurements.filter((m) => m.id !== id);
   224→    setMeasurements(updated);
   225→    saveMeasurements(updated);
   226→    toast.success("Запись удалена");
   227→  };
   228→
   229→  /* --- update personal best --- */
   230→  const savePB = () => {
   231→    const val = parseInt(pbInput);
   232→    if (isNaN(val) || val < 50 || val > 900) {
   233→      toast.error("Введите корректное значение (50-900)");
   234→      return;
   235→    }
   236→    const updated = { ...settings, personalBest: val };
   237→    setSettings(updated);
   238→    saveSettings(updated);
   239→    setEditingPB(false);
   240→    toast.success(`Персональный лучший результат: ${val} л/мин`);
   241→  };
   242→
   243→  /* --- notifications --- */
   244→  const scheduleReminder = () => {
   245→    const notifPref = loadNotificationsPref();
   246→    if (!notifPref) return;
   247→    if (reminderTimeout) clearTimeout(reminderTimeout);
   248→    const timeout = setTimeout(() => {
   249→      if (Notification.permission === "granted") {
   250→        new Notification("Пикфлоуметрия", {
   251→          body: "Прошло 30 минут — время сделать замер после ингаляции!",
   252→          tag: "peakflow-reminder",
   253→        });
   254→      }
   255→      toast.info("Прошло 30 минут — время сделать замер после ингаляции!", {
   256→        duration: 10000,
   257→      });
   258→    }, 30 * 60 * 1000); // 30 minutes
   259→    setReminderTimeout(timeout);
   260→  };
   261→
   262→  const toggleNotifications = async () => {
   263→    if (!notificationsEnabled) {
   264→      if ("Notification" in window) {
   265→        const perm = await Notification.requestPermission();
   266→        if (perm === "granted") {
   267→          setNotificationsEnabled(true);
   268→          saveNotificationsPref(true);
   269→          toast.success("Уведомления включены");
   270→        } else {
   271→          toast.error("Разрешите уведомления в настройках браузера");
   272→        }
   273→      } else {
   274→        toast.error("Уведомления не поддерживаются в этом браузере");
   275→      }
   276→    } else {
   277→      setNotificationsEnabled(false);
   278→      saveNotificationsPref(false);
   279→      if (reminderTimeout) clearTimeout(reminderTimeout);
   280→      toast.success("Уведомления выключены");
   281→    }
   282→  };
   283→
   284→  /* --- CSV export (fully client-side) --- */
   285→  const exportCSV = () => {
   286→    const sorted = [...measurements].sort(
   287→      (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime()
   288→    );
   289→
   290→    const pb = settings.personalBest;
   291→    const greenMin = round10(pb * 0.8);
   292→    const yellowMin = round10(pb * 0.5);
   293→
   294→    const bom = "\uFEFF";
   295→    const header = "Дата;Время;Период;Тип;ПСВ (л/мин);Зона\n";
   296→
   297→    const rows = sorted
   298→      .map((m) => {
   299→        const d = new Date(m.date);
   300→        const dateStr = d.toLocaleDateString("ru-RU", {
   301→          day: "2-digit",
   302→          month: "2-digit",
   303→          year: "numeric",
   304→        });
   305→        const timeStr = d.toLocaleTimeString("ru-RU", {
   306→          hour: "2-digit",
   307→          minute: "2-digit",
   308→        });
   309→        const periodStr = m.period === "morning" ? "Утро" : "Вечер";
   310→        const timingStr =
   311→          m.timing === "before" ? "До ингаляции" : "После ингаляции";
   312→        const zone =
   313→          m.value >= greenMin
   314→            ? "Зелёная"
   315→            : m.value >= yellowMin
   316→              ? "Жёлтая"
   317→              : "Красная";
   318→        return `${dateStr};${timeStr};${periodStr};${timingStr};${m.value};${zone}`;
   319→      })
   320→      .join("\n");
   321→
   322→    const csv = bom + header + rows;
   323→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   324→    const url = URL.createObjectURL(blob);
   325→    const a = document.createElement("a");
   326→    a.href = url;
   327→    a.download = `peakflow_${format(new Date(), "yyyy-MM-dd")}.csv`;
   328→    document.body.appendChild(a);
   329→    a.click();
   330→    document.body.removeChild(a);
   331→    URL.revokeObjectURL(url);
   332→    toast.success("CSV-файл загружен");
   333→  };
   334→
   335→  /* --- chart data --- */
   336→  const chartData = React.useMemo(() => {
   337→    const days = Array.from({ length: chartDays }, (_, i) => {
   338→      const d = startOfDay(subDays(new Date(), chartDays - 1 - i));
   339→      const dayStr = format(d, "yyyy-MM-dd");
   340→      const keys = [
   341→        { key: "morningBefore", period: "morning" as const, timing: "before" as const },
   342→        { key: "morningAfter", period: "morning" as const, timing: "after" as const },
   343→        { key: "eveningBefore", period: "evening" as const, timing: "before" as const },
   344→        { key: "eveningAfter", period: "evening" as const, timing: "after" as const },
   345→      ];
   346→
   347→      const point: Record<string, string | number> = {
   348→        date: format(d, "dd.MM", { locale: ru }),
   349→      };
   350→      for (const { key, period: p, timing: t } of keys) {
   351→        const m = measurements.find(
   352→          (m) =>
   353→            format(new Date(m.date), "yyyy-MM-dd") === dayStr &&
   354→            m.period === p &&
   355→            m.timing === t
   356→        );
   357→        point[key] = m ? m.value : (undefined as unknown as number);
   358→      }
   359→      return point;
   360→    });
   361→    return days;
   362→  }, [measurements, chartDays]);
   363→
   364→  const greenMin = round10(settings.personalBest * 0.8);
   365→  const yellowMin = round10(settings.personalBest * 0.5);
   366→
   367→  /* --- today's measurements --- */
   368→  const todayMeasurements = React.useMemo(() => {
   369→    const todayStr = format(new Date(), "yyyy-MM-dd");
   370→    return measurements.filter(
   371→      (m) => format(new Date(m.date), "yyyy-MM-dd") === todayStr
   372→    );
   373→  }, [measurements]);
   374→
   375→  const todayDone = (p: "morning" | "evening", t: "before" | "after") =>
   376→    todayMeasurements.some((m) => m.period === p && m.timing === t);
   377→
   378→  /* --- history list --- */
   379→  const historyList = React.useMemo(() => {
   380→    return [...measurements]
   381→      .sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime())
   382→      .slice(0, 20);
   383→  }, [measurements]);
   384→
   385→  if (!mounted) {
   386→    return (
   387→      <div className="min-h-screen bg-gradient-to-b from-slate-50 to-slate-100 flex items-center justify-center">
   388→        <Wind className="h-8 w-8 text-emerald-400 animate-pulse" />
   389→      </div>
   390→    );
   391→  }
   392→
   393→  return (
   394→    <div className="min-h-screen bg-gradient-to-b from-slate-50 to-slate-100 flex flex-col">
   395→      {/* Header */}
   396→      <header className="bg-white/80 backdrop-blur-sm border-b border-slate-200 sticky top-0 z-10">
   397→        <div className="max-w-lg mx-auto px-4 py-3 flex items-center justify-between">
   398→          <div className="flex items-center gap-2">
   399→            <Wind className="h-6 w-6 text-emerald-600" />
   400→            <h1 className="text-lg font-bold text-slate-800">Пикфлоуметрия</h1>
   401→          </div>
   402→          <div className="flex items-center gap-2">
   403→            <Button
   404→              variant="ghost"
   405→              size="icon"
   406→              onClick={toggleNotifications}
   407→              className="h-9 w-9"
   408→              title={
   409→                notificationsEnabled
   410→                  ? "Выключить уведомления"
   411→                  : "Включить уведомления"
   412→              }
   413→            >
   414→              {notificationsEnabled ? (
   415→                <Bell className="h-4 w-4 text-emerald-600" />
   416→              ) : (
   417→                <BellOff className="h-4 w-4 text-slate-400" />
   418→              )}
   419→            </Button>
   420→            <Button
   421→              variant="ghost"
   422→              size="icon"
   423→              onClick={exportCSV}
   424→              className="h-9 w-9"
   425→              title="Экспорт CSV"
   426→            >
   427→              <Download className="h-4 w-4 text-slate-600" />
   428→            </Button>
   429→          </div>
   430→        </div>
   431→      </header>
   432→
   433→      {/* Main content */}
   434→      <main className="flex-1 max-w-lg mx-auto w-full px-4 py-4 pb-8">
   435→        <Tabs defaultValue="input" className="w-full">
   436→          <TabsList className="w-full grid grid-cols-3">
   437→            <TabsTrigger value="input" className="gap-1.5">
   438→              <Plus className="h-3.5 w-3.5" />
   439→              <span className="text-xs sm:text-sm">Ввод</span>
   440→            </TabsTrigger>
   441→            <TabsTrigger value="chart" className="gap-1.5">
   442→              <Activity className="h-3.5 w-3.5" />
   443→              <span className="text-xs sm:text-sm">График</span>
   444→            </TabsTrigger>
   445→            <TabsTrigger value="history" className="gap-1.5">
   446→              <Clock className="h-3.5 w-3.5" />
   447→              <span className="text-xs sm:text-sm">История</span>
   448→            </TabsTrigger>
   449→          </TabsList>
   450→
   451→          {/* ===== INPUT TAB ===== */}
   452→          <TabsContent value="input">
   453→            <div className="space-y-4 mt-4">
   454→              {/* Today's checklist */}
   455→              <Card>
   456→                <CardHeader className="pb-2">
   457→                  <CardTitle className="text-sm font-medium flex items-center gap-2">
   458→                    <Check className="h-4 w-4 text-emerald-500" />
   459→                    Сегодня, {format(new Date(), "dd MMMM", { locale: ru })}
   460→                  </CardTitle>
   461→                </CardHeader>
   462→                <CardContent>
   463→                  <div className="grid grid-cols-2 gap-2">
   464→                    {(
   465→                      [
   466→                        { p: "morning" as const, t: "before" as const, label: "Утро (до)", icon: Sun },
   467→                        { p: "morning" as const, t: "after" as const, label: "Утро (после)", icon: Sun },
   468→                        { p: "evening" as const, t: "before" as const, label: "Вечер (до)", icon: Moon },
   469→                        { p: "evening" as const, t: "after" as const, label: "Вечер (после)", icon: Moon },
   470→                      ] as const
   471→                    ).map(({ p, t, label, icon: Icon }) => (
   472→                      <div
   473→                        key={`${p}-${t}`}
   474→                        className={`flex items-center gap-2 rounded-lg border p-2.5 text-sm transition-colors ${
   475→                          todayDone(p, t)
   476→                            ? "bg-emerald-50 border-emerald-200 text-emerald-700"
   477→                            : "bg-slate-50 border-slate-200 text-slate-500"
   478→                        } ${
   479→                          period === p && timing === t
   480→                            ? "ring-2 ring-emerald-500 ring-offset-1"
   481→                            : ""
   482→                        }`}
   483→                      >
   484→                        <Icon className="h-4 w-4 shrink-0" />
   485→                        <span className="text-xs font-medium leading-tight">
   486→                          {label}
   487→                        </span>
   488→                        {todayDone(p, t) && (
   489→                          <Check className="h-3.5 w-3.5 text-emerald-500 ml-auto shrink-0" />
   490→                        )}
   491→                      </div>
   492→                    ))}
   493→                  </div>
   494→                </CardContent>
   495→              </Card>
   496→
   497→              {/* Input card */}
   498→              <Card className="border-2 border-emerald-200">
   499→                <CardHeader className="pb-2">
   500→                  <CardTitle className="text-base font-semibold">
   501→                    Новый замер
   502→                  </CardTitle>
   503→                  <CardDescription>
   504→                    {period === "morning" ? (
   505→                      <span className="flex items-center gap-1">
   506→                        <Sun className="h-3.5 w-3.5 text-amber-500" /> Утро
   507→                      </span>
   508→                    ) : (
   509→                      <span className="flex items-center gap-1">
   510→                        <Moon className="h-3.5 w-3.5 text-indigo-400" /> Вечер
   511→                      </span>
   512→                    )}
   513→                    {" — "}
   514→                    {timing === "before"
   515→                      ? "до ингаляции"
   516→                      : "через 30 мин после ингаляции"}
   517→                  </CardDescription>
   518→                </CardHeader>
   519→                <CardContent className="space-y-4">
   520→                  {/* Period toggle */}
   521→                  <div className="flex gap-2">
   522→                    <Button
   523→                      variant={period === "morning" ? "default" : "outline"}
   524→                      size="sm"
   525→                      className="flex-1 gap-1.5"
   526→                      onClick={() => setPeriod("morning")}
   527→                    >
   528→                      <Sun className="h-3.5 w-3.5" /> Утро
   529→                    </Button>
   530→                    <Button
   531→                      variant={period === "evening" ? "default" : "outline"}
   532→                      size="sm"
   533→                      className="flex-1 gap-1.5"
   534→                      onClick={() => setPeriod("evening")}
   535→                    >
   536→                      <Moon className="h-3.5 w-3.5" /> Вечер
   537→                    </Button>
   538→                  </div>
   539→
   540→                  {/* Timing toggle */}
   541→                  <div className="flex gap-2">
   542→                    <Button
   543→                      variant={timing === "before" ? "default" : "outline"}
   544→                      size="sm"
   545→                      className="flex-1"
   546→                      onClick={() => setTiming("before")}
   547→                    >
   548→                      До ингаляции
   549→                    </Button>
   550→                    <Button
   551→                      variant={timing === "after" ? "default" : "outline"}
   552→                      size="sm"
   553→                      className="flex-1"
   554→                      onClick={() => setTiming("after")}
   555→                    >
   556→                      После ингаляции
   557→                    </Button>
